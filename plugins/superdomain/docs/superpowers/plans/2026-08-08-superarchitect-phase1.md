```python
from dataclasses import dataclass, field

CLASSIFICATIONS = {"core", "supporting", "generic"}
MODULE_LAYOUTS = {"multi-module", "single-module", "app-embedded"}
SHARED_ROLES = {"shared-kernel", "infrastructure", "client", "support"}
RELATION_TYPES = {"partnership", "customer-supplier", "conformist", "acl", "open-host", "published-language"}
PRESET_STYLES = {"layered-simple", "layered-domain", "hexagonal", "clean"}
KNOWN_PROFILES = {"kotlin-spring", "java-spring"}
TEMPLATE_MARKER = "<!-- superarchitect:template v1 -->"

@dataclass
class Application:
    name: str
    module_path: str
    contexts: list  # 컨텍스트명 목록 또는 ["all"]

@dataclass
class SharedModule:
    name: str
    path: str
    role: str

@dataclass
class Project:
    name: str
    line: int                 # 섹션 헤딩 라인 번호(1-기준)
    path: str = ""
    profile: str = ""
    base_package: str = ""
    test_location: str = ""
    applications: list = field(default_factory=list)    # [Application]
    shared_modules: list = field(default_factory=list)  # [SharedModule]
    package_conventions: dict = field(default_factory=dict)  # 레이어명 -> 패턴 문자열

@dataclass
class Module:
    name: str
    path: str
    layer: str

@dataclass
class Relation:
    target: str
    type: str
    contract: str

@dataclass
class RuleException:
    rule_id: str   # 예: "hex.ports-owned-inside" ("-" 접두 제거 후)
    adr: str       # 예: "ADR-0002" (없으면 "")

@dataclass
class Context:
    name: str
    line: int
    project: str = ""         # 프로젝트 1개면 파서가 자동 귀속
    classification: str = ""
    style: str = ""
    module_layout: str = ""
    patterns: list = field(default_factory=list)         # [str]
    rule_exceptions: list = field(default_factory=list)  # [RuleException]
    transition: tuple = ()    # (출발, 목표) — 파싱 성공(→ 존재) 시에만
    transition_raw: str = "" # 이행 라벨이 존재하면 원문 값을 그대로 보존(성공/실패 무관).
                              # ""=라벨 없음, 값 있음+transition=()는 형식 오류(Task 3 판정)
    modules: list = field(default_factory=list)          # [Module]
    relations: list = field(default_factory=list)        # [Relation]

@dataclass
class ParseError:
    line: int      # 0 = 문서 전체 수준 오류
    message: str   # 한국어

@dataclass
class Architecture:
    projects: list = field(default_factory=list)
    contexts: list = field(default_factory=list)
    has_template_marker: bool = False

def parse_architecture(text: str) -> tuple:
    """(Architecture, [ParseError])를 반환. 이 태스크는 구조 추출만 수행하며
    오류 목록은 항상 []다 — 필수 결정 누락·비정규 값 검증(Task 3)은 이후
    validate(architecture) 호출을 여기에 추가해 채운다."""
```

라인 기반 상태 머신으로 구현한다. 핵심 규칙:

```python
import re

RE_PROJECT = re.compile(r"^##\s+프로젝트:\s*(.+?)\s*$")
RE_CONTEXT = re.compile(r"^##\s+컨텍스트:\s*(.+?)\s*$")
RE_H2      = re.compile(r"^##\s+")
RE_H3      = re.compile(r"^###\s+(.+?)\s*$")
RE_LABEL   = re.compile(r"^-\s*(.+?)\s*:\s*(.+?)\s*$")
RE_TABLE   = re.compile(r"^\|")

def _strip_comment(value: str) -> str:
    """후행 '(...)' 주석 제거. 예: 'core (core | supporting)' -> 'core'"""
    return re.sub(r"\s*\([^)]*\)\s*$", "", value).strip()
```

- 상태: 현재 섹션(project | context | None)과 현재 소제목(`###` 이름 또는 None)을 추적한다.
- `## 프로젝트:`/`## 컨텍스트:` 매치 → 새 섹션 시작(라인 번호 기록). 그 외 `##` 헤딩 → 섹션 없음(무시 모드).
- 섹션 내 `- 키: 값` → 알려진 라벨만 처리, 모르는 키는 무시. `규칙 예외`를 제외한 모든 값에 `_strip_comment` 적용.
  - `패턴`: 쉼표 분리 리스트.
  - `규칙 예외`: 쉼표 분리 후 각 항목을 `^-?\s*([A-Za-z0-9_.\-]+)\s*(?:\((ADR-\d{4})\))?$`로 파싱 → `RuleException(rule_id, adr)`. 정규식이 매치하지 않아도 항목을 버리지 않는다 — 앞의 `-`만 제거한 원문을 `rule_id`로, `adr=""`로 보존해 Task 3이 형식 오류로 판정할 수 있게 한다.
  - `이행`: `→` 기준 분리 → `(출발, 목표)`. 라벨 값은 성공 여부와 무관하게 `transition_raw`에 그대로 보존한다. `→`가 없으면 `transition`은 `()`로 남고(검증 오류는 Task 3), `transition_raw`로 "라벨 없음"과 "형식 오류"를 구분할 수 있다.
- 표 파싱: `|` 라인 연속 블록을 (헤더, 구분, 데이터행들)로 읽는다. 셀은 `|` 분리 후 strip. 표의 의미는 위치로 결정:
  - 프로젝트 섹션 + 소제목 `애플리케이션` → Application 행 (이름/모듈 경로/포함 컨텍스트, 포함 컨텍스트는 쉼표 분리)
  - 프로젝트 섹션 + 소제목 `공용 모듈` → SharedModule 행
  - 프로젝트 섹션 + 소제목 `패키지 규약` → package_conventions[레이어] = 패턴 (패턴 셀의 후행 주석은 유지하지 않도록 `_strip_comment` 대신 셀 그대로 사용하되 표 뒤 화살표 주석 행은 없다고 가정)
  - 컨텍스트 섹션 + 소제목 없음 + 첫 표 → Module 행
  - 컨텍스트 섹션 + 소제목 `관계` → Relation 행
  - 그 외 표는 무시.
- 컨텍스트의 `프로젝트` 라벨이 비어 있고 프로젝트가 정확히 1개면 그 이름으로 자동 귀속(파싱 직후 후처리).
- `has_template_marker`: 문서 어느 라인이든 `TEMPLATE_MARKER` 포함 여부.

```python
class TestNormalize(unittest.TestCase):
    def test_multi_module_default_convention(self):
        arch, _ = parse_architecture(load("minimal.md"))
        pats = normalize(arch)
        got = {(p.context, p.layer, p.pattern) for p in pats}
        self.assertIn(("claim", "domain", "com.acme.claim.domain.."), got)
        self.assertIn(("claim", "application", "com.acme.claim.application.."), got)
        self.assertIn(("claim", "adapter", "com.acme.claim.adapter.."), got)
        self.assertEqual(len([p for p in pats if p.layer == "adapter"]), 1)  # in/out 중복 제거

    def test_app_embedded_conventions(self):
        arch, _ = parse_architecture(load("full.md"))
        pats = normalize(arch)
        got = {(p.context, p.layer, p.pattern) for p in pats}
        self.assertIn(("brawlstars", "domain", "com.imstargg.core.domain.brawlstars.."), got)
        self.assertIn(("brawlstars", "application", "com.imstargg.core.application.brawlstars.."), got)
        # {앱} 패턴은 앱별 전개(하이픈→점 치환), 컨텍스트 비분할이므로 context="*"
        self.assertIn(("*", "presentation", "com.imstargg.core.api.."), got)
        self.assertIn(("*", "presentation", "com.imstargg.core.batch.."), got)

    def test_single_module_all_layer(self):
        text = load("minimal.md").replace("- 모듈 구성: multi-module", "- 모듈 구성: single-module") \
            .replace("| claim-domain | claim/domain | domain |", "| claim | claim | all |")
        # 나머지 모듈 행 3개 삭제
        for row in ["| claim-application | claim/application | application |",
                    "| claim-adapter-in | claim/adapter-in | adapter |",
                    "| claim-adapter-out | claim/adapter-out | adapter |"]:
            text = text.replace(row + "\n", "")
        arch, errors = parse_architecture(text)
        self.assertEqual(errors, [])
        pats = normalize(arch)
        self.assertIn(("claim", "all", "com.acme.claim.."), {(p.context, p.layer, p.pattern) for p in pats})
```

```markdown
1. 입력을 파악한다: 대화 내용, URL(WebFetch로 읽기), 붙여넣은 노트.
2. `references/INDEX.md`를 읽어 대상을 결정한다:
   - 입력이 기존 문서(key·summary가 같은 주제)와 겹치면 → **보완 모드**
   - 겹치는 문서가 없으면 → **신규 모드**
   - 판단이 애매하면 후보 문서를 제시하고 사용자에게 확인한다.
3. 신규 모드: `references/governance/knowledge-doc-template.md` 형식으로 초안을 작성한다 —
   frontmatter의 summary는 필수. 입력이 충분하면 "## 적용 기준"과 "## 규칙" 섹션까지
   (성숙 문서), 부족하면 자유 형식 본문만(draft — 정상 상태다. 완벽한 구조보다 일단
   넣는 것이 우선). 주제는 `references/knowledge/` 아래 기존 주제 디렉터리에서 제안하고,
   맞는 것이 없으면 새 디렉터리를 만든다(주제 아래 추가 중첩 금지). 파일명(=key)은
   kebab-case 영어로 짓는다.
4. 보완 모드: 기존 문서를 읽고 새 내용을 병합한다 — 기존 구조·서술을 유지하며 통합하고,
   중복은 합치고, 기존 내용과 모순되면 임의로 대체하지 않고 어느 쪽이 맞는지 사용자에게
   확인한다. draft 스텁에 "## 적용 기준"·"## 규칙"을 갖춰주면 성숙으로 승격된다.
   문서가 500줄을 넘으면 분할을 제안한다.
5. 다른 문서와 연결되면 본문에 [[key]]로 참조한다.
6. `python3 scripts/build_index.py`를 실행한다. 오류가 나면 고치고 재실행한다.
7. INDEX.md의 해당 행(draft 변화 포함)과 변경 요약을 보여주고 커밋을 제안한다:
   `git add references/ && git commit -m "knowledge: <key> 추가|보완"`

## 주의

- INDEX.md를 직접 편집하지 않는다(생성물).
- 내용을 창작하지 않는다 — 입력에 없는 주장은 넣지 않고, 불확실하면 사용자에게 확인한다.
- 보완 시 기존 문서의 올바른 내용을 삭제하지 않는다 — 대체가 필요하면 근거를 확인받는다.
```

### Task 12: init 스킬

**Files:**
- Create: `skills/init/SKILL.md`

**Interfaces:**
- Consumes: parse_architecture.py CLI(검증), architecture-template.md(스켈레톤·라벨 사전), knowledge 5종(Task 8), adr-template.md.
- Produces: `/superarchitect:init` — ARCHITECTURE.md·docs/architecture/summary.md·decisions/0001을 생성하는 스킬. summary.md 규격(아래)은 이후 모든 파생물 갱신 스킬이 따른다.

- [ ] **Step 1: SKILL.md 작성**

frontmatter description은 스펙 §6의 예시를 그대로 쓴다. 본문은 스펙 §6.1의 9단계를 절차로 옮기되, 다음을 정확히 포함한다 (500줄 이하):

1. **재초기화 확인** — 루트에 ARCHITECTURE.md가 있으면 덮어쓸지 확인.
2. **후보 감지** — git 루트에서 빌드 파일 스캔: `settings.gradle(.kts)`·`pom.xml`(JVM), `package.json`(Node), `go.mod`(Go). 제안 기준은 **프로파일 매칭**(현재 kotlin-spring·java-spring → JVM만 제안). 프로파일 없는 후보는 "감지됨 — 대응 프로파일 없음, 범위 외"로 표시. 스택 추측은 안내 문구에만 사용, 제외 판정 금지. 최종 확정은 사용자.
3. **후보 0개(그린필드)** — 신규 설계 인터뷰: 문제 한 문단 → 용어가 다른 업무 영역 → 함께/따로 바뀌는 것 → 팀·배포 단위. 한 번에 하나씩 질문. 프로젝트 경로는 예정 경로로 선언. 도메인이 흐릿하면 `/superarchitect:model` 스토밍 모드 먼저 제안.
4. **기존 코드 스캔** — 모듈 구조·패키지에서 컨텍스트 후보와 실현 형태(모듈 구성 3형 중 판별: 레이어별 모듈이면 multi-module, 컨텍스트별 단일 모듈이면 single-module, 실행 앱 모듈들 + 레이어-우선 패키지면 app-embedded) 추정 제시. 멀티 앱이면 애플리케이션·공용 모듈·패키지 규약 표 도출.
5. **분류·스타일 결정** — 이 시점에 `references/INDEX.md`를 읽고 `domain-classification`, `bounded-contexts`, `module-composition`을 읽어 체크리스트로 질문. 분류→스타일 기본값 제안, 최종 결정은 사용자. 프리셋이 안 맞으면 커스텀 스타일 선언 절차 안내(architecture-template.md의 선언 형식).
6. **ARCHITECTURE.md 작성** — architecture-template.md의 스켈레톤 사용. 기존 프로젝트에서 현재≠목표면 이행 필드 기록(baseline 동결은 Phase 2의 fitness 이후 가능하므로 v1 Phase 1에서는 "이행 선언만 하고 baseline은 fitness 구현 후 생성됨"을 안내). 작성 후 반드시 `python3 ${CLAUDE_PLUGIN_ROOT}/scripts/parse_architecture.py ARCHITECTURE.md` 실행 — 통과할 때까지 수정(실패 상태로 종료 금지).
7. **파생물 생성** — `docs/architecture/summary.md`(규격: 파생물 공통 헤더 + 30줄 이내 + 프로젝트·컨텍스트 표(이름/분류/스타일/모듈 수) + 전역 핵심 규칙 3~5개 + 마지막 줄 `상세: ARCHITECTURE.md · 검토: /superarchitect:review`), 컨텍스트 맵 생성 구역(`<!-- superarchitect:generated:context-map -->` ~ `<!-- /superarchitect:generated -->` 사이 mermaid graph — 컨텍스트 노드·관계 엣지·애플리케이션 노드), `docs/architecture/decisions/0001-record-architecture-decisions.md`(adr-template 형식, status: accepted), 초기 주요 결정(스타일 선택 등)은 0002+로.
8. **마무리 제안** — CLAUDE.md에 "아키텍처 기준: ARCHITECTURE.md 참조" 한 줄 추가 제안(선택), git pre-commit 훅 설치 제안은 "check 스크립트가 갖춰지는 Phase 2 이후 제공"임을 안내, `/superarchitect:fitness` 실행 권유(Phase 2 전이면 "아직 미구현"을 안내).

또한 "지식 참조 프로토콜" 명시: knowledge를 디렉터리째 읽지 말 것, INDEX 경유 선별 로드만.
